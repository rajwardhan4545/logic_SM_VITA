public class demo {

		static void check(int []arr,int n)
		{
			int max1=arr[0];
			int max2=Integer.MIN_VALUE;
			
			int min1=arr[0];
			int min2=Integer.MAX_VALUE;
			
			for(int i=1;i<n;i++)
			{
				if(arr[i]>max1)
				{
					max2=max1;
					max1=arr[i];
				}
				if(arr[i]<min1)
				{
					min2=min1;
					min1=arr[i];
				}
			}
			
			System.out.println(max2+" "+max1);
			System.out.println(min2+" "+min1);
			
		}
	 
	 
		public static void main(String[] args) {
			
			int []arr= {0,-3,-4,6,1,7,8,-9,0,10};
			int n=arr.length;		
			check(arr,n);
		}


}
